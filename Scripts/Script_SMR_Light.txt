CREATE PROCEDURE SP_LISTAR_CONFIRMACAO_MEDICAO
(
@Usuario varchar(10)
)
AS

SELECT Log_Tarefas.Cod_Usuario, 
	Usuarios.Nome_Usuario AS Nome, 
	Log_Tarefas.Data_Atualizacao, 
	CONVERT(varchar(10),Log_Tarefas.Data_Atualizacao,103) as Data,
	MSP_PROJECTS.PROJ_NAME, 
	Log_Tarefas.TASK_UID,
	MSP_TASKS.TASK_NAME,
	CONVERT(varchar(10),MSP_TASKS.TASK_START_DATE,103) as TASK_START_DATE, 
	CONVERT(varchar(10),MSP_TASKS.TASK_FINISH_DATE,103) as TASK_FINISH_DATE, 
	Log_Tarefas.TASK_PCT_COMP
FROM Log_Tarefas INNER JOIN Usuarios 
	ON Log_Tarefas.Cod_Usuario = Usuarios.Login COLLATE Latin1_General_CI_AS 
	INNER JOIN MSP_PROJECTS ON Log_Tarefas.PROJ_ID = MSP_PROJECTS.PROJ_ID 
	INNER JOIN MSP_TASKS ON Log_Tarefas.TASK_UID = MSP_TASKS.TASK_UID AND Log_Tarefas.PROJ_ID = MSP_TASKS.PROJ_ID 
Where CONVERT(VARCHAR(8),Data_Atualizacao,112) = CONVERT(VARCHAR(8),getdate(),112) AND 
	UPPER(RTRIM(LTRIM(Log_Tarefas.Cod_Usuario))) = UPPER(RTRIM(LTRIM(@Usuario)))
Order By MSP_PROJECTS.PROJ_NAME, Log_Tarefas.Data_Atualizacao desc

	TK.TASK_PCT_COMP <> 100 AND
	TK.PROJ_ID = 1

		TK.TASK_START_DATE, 
		TK.TASK_FINISH_DATE, 
		TK.TASK_PCT_COMP, 
		TK.TASK_IS_SUMMARY, 
		TK.TASK_IS_EXTERNAL, 
		TK.PROJ_ID, 
		PJ.PROJ_NAME, 
		TK.TASK_UID, 
		TK.TASK_ID, 
		TK.TASK_DUR, 
		UPPER(DC.TEXT_VALUE) as CLI, 
		Null as Frente, 
		Null as Equipe, 
		Null as Governanca, 
		Null as RespInt, 
		Null as  Area,
		Null as  Unidade 
	FROM MSP_PROJECTS PJ, MSP_TASKS TK left outer join DadosCLI DC on 
		TK.PROJ_ID = DC.PROJ_ID and TK.TASK_UID = DC.TEXT_REF_UID 
	WHERE PJ.PROJ_ID = TK.PROJ_ID 
 		And (TK.TASK_ID IS NOT NULL)  
		And (TK.TASK_ID <> 0) 
		And (TK.TASK_IS_SUMMARY = 1) 
	Order By TK.PROJ_ID, TK.TASK_OUTLINE_NUM

	SELECT * FROM Usuarios
